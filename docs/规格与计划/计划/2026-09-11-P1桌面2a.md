# OpenField P1 — Plan 2a: 桌面端地基（Vault + 证据入库）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立 `apps/desktop` Electron 应用骨架，实现 SQLCipher 加密 vault、Schema v1 迁移、证据链持久化、实体仓库与 Ingest 服务（单文件导入 + 手机 bundle 导入）——全部 vitest 可测，无 UI。

```ts
import DatabaseConstructor from 'better-sqlite3-multiple-ciphers';

export type Db = InstanceType<typeof DatabaseConstructor>;

// 口令由 SQLCipher 自行做 PBKDF2 派生（PRAGMA key 传口令字符串语义）。
// 外键约束必须显式开启（SQLite 默认关闭）——证据系统不接受孤儿记录。
export function openDb(path: string, passphrase: string): Db {
  const db = new DatabaseConstructor(path);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');
  db.pragma(`key = '${passphrase.replace(/'/g, "''")}'`);
  return db;
}
```

```ts
import type { Db } from './db';

export const SCHEMA_VERSION = 1;

const V1_STATEMENTS: string[] = [
  `CREATE TABLE evidence_log (
    seq INTEGER PRIMARY KEY,
    ts INTEGER NOT NULL,
    actor TEXT NOT NULL,
    action TEXT NOT NULL,
    payload_hash TEXT NOT NULL,
    prev_hash TEXT NOT NULL,
    entry_hash TEXT NOT NULL UNIQUE
  )`,
  `CREATE TRIGGER evidence_log_no_update BEFORE UPDATE ON evidence_log
    BEGIN SELECT RAISE(ABORT, 'evidence_log is append-only'); END`,
  `CREATE TRIGGER evidence_log_no_delete BEFORE DELETE ON evidence_log
    BEGIN SELECT RAISE(ABORT, 'evidence_log is append-only'); END`,
  `CREATE TABLE field_event (
    id TEXT PRIMARY KEY,
    date TEXT NOT NULL,
    city_code TEXT NOT NULL,
    location_name TEXT NOT NULL,
    gps_lat REAL,
    gps_lng REAL,
    context_note TEXT,
    created_seq INTEGER NOT NULL REFERENCES evidence_log(seq)
  )`,
  `CREATE TABLE encounter (
    id TEXT PRIMARY KEY,
    event_id TEXT NOT NULL REFERENCES field_event(id),
    participant_ref TEXT NOT NULL,
    sampling_reason TEXT NOT NULL,
    consent_record_id TEXT,
    started_at INTEGER NOT NULL,
    note TEXT,
    created_seq INTEGER NOT NULL REFERENCES evidence_log(seq)
  )`,
  `CREATE INDEX idx_encounter_event ON encounter(event_id)`,
  `CREATE TABLE participant (
    pseudonym TEXT PRIMARY KEY,
    industry TEXT,
    region TEXT,
    referral_chain TEXT,
    consent_scope TEXT,
    created_seq INTEGER NOT NULL REFERENCES evidence_log(seq)
  )`,
  `CREATE TABLE participant_identity (
    pseudonym TEXT PRIMARY KEY REFERENCES participant(pseudonym),
    real_name TEXT NOT NULL,
    created_seq INTEGER NOT NULL REFERENCES evidence_log(seq)
  )`,
  `CREATE TABLE artifact (
    id TEXT PRIMARY KEY,
    encounter_id TEXT REFERENCES encounter(id),
    event_id TEXT REFERENCES field_event(id),
    type TEXT NOT NULL,
    sha256 TEXT NOT NULL UNIQUE,
    size INTEGER NOT NULL,
    mime TEXT NOT NULL,
    captured_at INTEGER NOT NULL,
    gps_lat REAL,
    gps_lng REAL,
    device_id TEXT NOT NULL,
    version INTEGER NOT NULL DEFAULT 1,
    ref_id TEXT,
    originals_path TEXT NOT NULL,
    ingested_seq INTEGER NOT NULL REFERENCES evidence_log(seq)
  )`,
  `CREATE INDEX idx_artifact_event ON artifact(event_id)`,
  `CREATE INDEX idx_artifact_encounter ON artifact(encounter_id)`,
  `CREATE TABLE consent_record (
    id TEXT PRIMARY KEY,
    encounter_id TEXT NOT NULL REFERENCES encounter(id),
    template_type TEXT NOT NULL,
    signature_artifact_id TEXT,
    verbal_consent_artifact_id TEXT,
    scope TEXT NOT NULL,
    withdrawn_at INTEGER,
    created_seq INTEGER NOT NULL REFERENCES evidence_log(seq)
  )`,
  `CREATE TABLE memo (
    id TEXT PRIMARY KEY,
    linked_artifact_ids TEXT NOT NULL DEFAULT '[]',
    type TEXT NOT NULL,
    content TEXT NOT NULL,
    created_at INTEGER NOT NULL,
    confirmed_at INTEGER,
    confirmed_seq INTEGER REFERENCES evidence_log(seq)
  )`,
  `CREATE TABLE inbox_item (
    id TEXT PRIMARY KEY,
    source_path TEXT NOT NULL,
    detected_at INTEGER NOT NULL,
    sha256 TEXT,
    suggested_event_id TEXT,
    suggested_encounter_id TEXT,
    status TEXT NOT NULL DEFAULT 'pending'
  )`,
  `CREATE INDEX idx_inbox_status ON inbox_item(status)`,
  `CREATE TABLE time_sync (
    id TEXT PRIMARY KEY,
    checked_at INTEGER NOT NULL,
    ntp_server TEXT NOT NULL,
    offset_ms INTEGER NOT NULL,
    logged_seq INTEGER NOT NULL REFERENCES evidence_log(seq)
  )`,
];

export interface Migration {
  version: number;
  statements: string[];
}

export const MIGRATIONS: Migration[] = [{ version: 1, statements: V1_STATEMENTS }];

export function migrate(db: Db): void {
  const current = db.pragma('user_version', { simple: true }) as number;
  for (const m of MIGRATIONS) {
    if (m.version > current) {
      db.transaction(() => {
        for (const s of m.statements) db.exec(s);
        db.pragma(`user_version = ${m.version}`);
      })();
    }
  }
}
```

```ts
import DatabaseConstructor from 'better-sqlite3-multiple-ciphers';
import { computePayloadHash } from '@openfield/core';
import { GENESIS_PREV_HASH, verifyChain } from '@openfield/core';
import { appendWithinTx, getHead, getAllEntries, verifyAll } from '../src/main/services/evidence-repo';

// 无触发器的裸表：用于制造"已绕过应用层"的篡改场景，检验 verifyAll 的检出能力
function bareDb() {
  const db = new DatabaseConstructor(':memory:');
  db.exec(`CREATE TABLE evidence_log (
    seq INTEGER PRIMARY KEY, ts INTEGER NOT NULL, actor TEXT NOT NULL, action TEXT NOT NULL,
    payload_hash TEXT NOT NULL, prev_hash TEXT NOT NULL, entry_hash TEXT NOT NULL UNIQUE)`);
  return db;
}

describe('appendWithinTx', () => {
  it('首条 seq=0、prevHash=GENESIS；后续接续', () => {
    const db = bareDb();
    const e1 = appendWithinTx(db, { ts: 1, actor: 'desktop', action: 'CREATE_EVENT', payloadHash: computePayloadHash({ n: 1 }) });
    expect(e1.seq).toBe(0);
    expect(e1.prevHash).toBe(GENESIS_PREV_HASH);
    const e2 = appendWithinTx(db, { ts: 2, actor: 'desktop', action: 'INGEST_ARTIFACT', payloadHash: computePayloadHash({ n: 2 }) });
    expect(e2.seq).toBe(1);
    expect(e2.prevHash).toBe(e1.entryHash);
    db.close();
  });

  it('getHead 返回最新条目；空链返回 null', () => {
    const db = bareDb();
    expect(getHead(db)).toBeNull();
    appendWithinTx(db, { ts: 1, actor: 'desktop', action: 'EXPORT', payloadHash: computePayloadHash({}) });
    expect(getHead(db)?.seq).toBe(0);
    db.close();
  });
});

describe('verifyAll', () => {
  it('完整链通过', () => {
    const db = bareDb();
    for (let i = 0; i < 3; i++) {
      appendWithinTx(db, { ts: i, actor: 'desktop', action: 'INGEST_ARTIFACT', payloadHash: computePayloadHash({ i }) });
    }
    expect(verifyAll(db)).toEqual({ ok: true });
    db.close();
  });

  it('绕过应用层篡改 payload_hash → 检出', () => {
    const db = bareDb();

      appendWithinTx(db, { ts: i, actor: 'desktop', action: 'INGEST_ARTIFACT', payloadHash: computePayloadHash({ i }) });

    const r = verifyAll(db);
    expect(r.ok).toBe(false);
    expect(r.ok === false && r.brokenAt).toBe(1);
    db.close();
```

```ts
import { computeEntryHash, GENESIS_PREV_HASH, verifyChain } from '@openfield/core';
import type { EvidenceAction, EvidenceEntry } from '@openfield/core';
import type { Db } from '../db/db';

export interface EntryInput {
  ts: number;
  actor: string;
  action: EvidenceAction;
  payloadHash: string;
}

export function getHead(db: Db): { seq: number; entryHash: string } | null {
  const row = db
    .prepare('SELECT seq, entry_hash AS entryHash FROM evidence_log ORDER BY seq DESC LIMIT 1')
    .get() as { seq: number; entryHash: string } | undefined;
  return row ?? null;
}

// 必须在调用方的 db.transaction 内使用：读链头→构造→插入，保证原子性。
export function appendWithinTx(db: Db, input: EntryInput): EvidenceEntry {
  const head = getHead(db);
  const candidate = {
    seq: head ? head.seq + 1 : 0,
    ts: input.ts,
    actor: input.actor,
    action: input.action,
    payloadHash: input.payloadHash,
    prevHash: head ? head.entryHash : GENESIS_PREV_HASH,
  };
  const entry: EvidenceEntry = { ...candidate, entryHash: computeEntryHash(candidate) };
  db.prepare(
    'INSERT INTO evidence_log (seq, ts, actor, action, payload_hash, prev_hash, entry_hash) VALUES (?, ?, ?, ?, ?, ?, ?)',
  ).run(entry.seq, entry.ts, entry.actor, entry.action, entry.payloadHash, entry.prevHash, entry.entryHash);
  return entry;
}

export function getAllEntries(db: Db): EvidenceEntry[] {
  const rows = db
    .prepare(
      'SELECT seq, ts, actor, action, payload_hash AS payloadHash, prev_hash AS prevHash, entry_hash AS entryHash FROM evidence_log ORDER BY seq ASC',
    )
    .all() as unknown[];
  return rows.map((r) => EvidenceEntry.parse(r));
}

export function verifyAll(db: Db) {
  return verifyChain(getAllEntries(db));
}
```

```ts
import { existsSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';
import { openDb } from '../db/db';
import type { Db } from '../db/db';
import { migrate } from '../db/migrations';

export class VaultKeyError extends Error {}
export class VaultExistsError extends Error {}

export interface VaultPaths {
  root: string;
  dbPath: string;
  originalsDir: string;
  backupsDir: string;
}

export function vaultLayout(root: string): VaultPaths {
  return {
    root,
    dbPath: join(root, 'vault.db'),
    originalsDir: join(root, 'originals'),
    backupsDir: join(root, 'backups'),
  };
}

export function createVault(root: string, passphrase: string): { db: Db; paths: VaultPaths } {
  const paths = vaultLayout(root);
  if (existsSync(paths.dbPath)) throw new VaultExistsError(`vault 已存在：${paths.dbPath}`);
  mkdirSync(paths.originalsDir, { recursive: true });
  mkdirSync(paths.backupsDir, { recursive: true });
  const db = openDb(paths.dbPath, passphrase);
  try {
    migrate(db);
  } catch (e) {
    db.close();
    throw new VaultKeyError(`vault 初始化失败：${String(e)}`);
  }
  return { db, paths };
}

export function openVault(root: string, passphrase: string): { db: Db; paths: VaultPaths } {
  const paths = vaultLayout(root);
  if (!existsSync(paths.dbPath)) throw new Error(`vault 不存在：${paths.dbPath}`);
  const db = openDb(paths.dbPath, passphrase);
  try {
    // 口令错误时首个真实读页即抛 SQLITE_NOTADB；用 sqlite_master 探测
    db.prepare('SELECT count(*) AS n FROM sqlite_master').get();
    migrate(db);
  } catch {
    db.close();
    throw new VaultKeyError('口令错误或 vault 已损坏');
  }
  return { db, paths };
}

export function lockVault(db: Db): void {
  db.close();
}
```

```ts
  '.m4a': 'audio/mp4', '.mp3': 'audio/mpeg', '.wav': 'audio/wav', '.aac': 'audio/aac',
  '.heic': 'image/heic', '.jpg': 'image/jpeg', '.jpeg': 'image/jpeg', '.png': 'image/png',

function artifactTypeFromMime(mime: string): ArtifactType {
  if (mime.startsWith('audio/')) return 'audio';
  if (mime.startsWith('image/')) return 'photo';
  if (mime === 'text/plain' || mime === 'text/markdown') return 'note';
  return 'doc';
}

export async function sha256File(path: string): Promise<string> {
  const hash = createHash('sha256');
  await pipeline(createReadStream(path), hash);
  return hash.digest('hex');
}

interface IngestArtifactArgs {
  sourcePath: string;
  sha256: string;
  size: number;
  mime: string;
  type: ArtifactType;
  capturedAt: number;
  gps?: Gps;
  deviceId: string;
  eventId?: string;
  encounterId?: string;
  actor: string;

// 必须在 db.transaction 内调用。返回 null 表示同 sha256 已登记（幂等跳过）。
function ingestArtifactCore(db: Db, paths: VaultPaths, a: IngestArtifactArgs): { artifactId: string; seq: number } | null {
  if (getArtifactBySha256(db, a.sha256)) return null;
  const artifactId = `art-${randomUUID()}`;
  const ext = extname(a.sourcePath).toLowerCase();
  const relPath = `originals/${artifactId}/v1${ext}`;
  const destDir = join(paths.originalsDir, artifactId);
  mkdirSync(destDir, { recursive: true });
  const tmpPath = join(destDir, `.tmp-${randomUUID()}`);
  copyFileSync(a.sourcePath, tmpPath);
  renameSync(tmpPath, join(destDir, `v1${ext}`));
  chmodSync(join(destDir, `v1${ext}`), 0o444);

  const entry = appendWithinTx(db, {
    ts: Date.now(),
    actor: a.actor,
    action: 'INGEST_ARTIFACT',
    payloadHash: computePayloadHash({ artifactId, sha256: a.sha256, size: a.size, type: a.type }),
  });

  const artifact: Parameters<typeof insertArtifact>[1] = {
    id: artifactId,
    encounterId: a.encounterId,
    eventId: a.eventId,
    type: a.type,
    sha256: a.sha256,
    size: a.size,
    mime: a.mime,
    capturedAt: a.capturedAt,
    gps: a.gps,
    deviceId: a.deviceId,
    version: 1,
  };
  insertArtifact(db, artifact, relPath, entry.seq);
  return { artifactId, seq: entry.seq };
}

export interface IngestFileInput {
  sourcePath: string;
  eventId?: string;
  encounterId?: string;
  actor?: string;
  type?: ArtifactType;
  capturedAt?: number;
  deviceId?: string;
}

export type IngestOutcome = { status: 'ingested' | 'duplicate'; artifactId: string; sha256: string };

// 独立文件导入：无 bundle 元数据，capturedAt 退化为 mtime（EXIF/GPS 提取在 Plan 2b InboxWatcher 一并处理）
export async function ingestFile(db: Db, paths: VaultPaths, input: IngestFileInput): Promise<IngestOutcome> {
  const sha256 = await sha256File(input.sourcePath);
  const existing = getArtifactBySha256(db, sha256);
  if (existing) return { status: 'duplicate', artifactId: existing.id, sha256 };
  const mime = mimeFromExt(extname(input.sourcePath));
  const args: IngestArtifactArgs = {
    sourcePath: input.sourcePath,
    sha256,
    size: statSync(input.sourcePath).size,
    mime,
    type: input.type ?? artifactTypeFromMime(mime),
    capturedAt: input.capturedAt ?? statSync(input.sourcePath).mtimeMs,
    deviceId: input.deviceId ?? hostname(),
    eventId: input.eventId,
    encounterId: input.encounterId,
    actor: input.actor ?? 'desktop',
  };
  return db.transaction(() => {
    const r = ingestArtifactCore(db, paths, args);
    return { status: 'ingested' as const, artifactId: r!.artifactId, sha256 };
  })();
}

  encountersCreated: number;

  memosStored: number;
  artifactsIngested: string[];

// 媒体哈希（异步 IO）在事务外完成；事务内只做同步 DB 操作
export async function ingestBundle(db: Db, paths: VaultPaths, input: IngestBundleInput): Promise<IngestBundleOutcome> {
  const { bundle, mediaDir, actor = 'desktop' } = input;
  const resolved: Array<{ ref: MediaRef; sourcePath?: string; actualSha?: string }> = [];
  for (const ref of bundle.mediaRefs) {
    if (!mediaDir) {
      resolved.push({ ref });
      continue;
    }
    const p = join(mediaDir, ref.filename);
    if (!existsSync(p)) {
      resolved.push({ ref });
      continue;
    }
    resolved.push({ ref, sourcePath: p, actualSha: await sha256File(p) });
  }

  return db.transaction(() => {
    const outcome: IngestBundleOutcome = {
      eventsCreated: 0, encountersCreated: 0, participantsCreated: 0,
      consentsRecorded: 0, memosStored: 0,
      artifactsIngested: [], pendingMedia: [], quarantinedMedia: [],
    };
    const now = Date.now();

    for (const ev of bundle.events) {
      if (existsById(db, 'field_event', ev.id)) continue;
      const entry = appendWithinTx(db, { ts: now, actor, action: 'CREATE_EVENT', payloadHash: computePayloadHash(ev) });
      insertFieldEvent(db, ev, entry.seq);
      outcome.eventsCreated++;
    }
    for (const enc of bundle.encounters) {
      if (existsById(db, 'encounter', enc.id)) continue;
      const entry = appendWithinTx(db, { ts: now, actor, action: 'CREATE_ENCOUNTER', payloadHash: computePayloadHash(enc) });
      insertEncounter(db, enc, entry.seq);
      outcome.encountersCreated++;
    }
    for (const p of bundle.participants) {
      if (existsParticipant(db, p.pseudonym)) continue;
      const entry = appendWithinTx(db, { ts: now, actor, action: 'CREATE_PARTICIPANT', payloadHash: computePayloadHash(p) });
      insertParticipant(db, p, entry.seq);
      outcome.participantsCreated++;
    }
    for (const c of bundle.consents) {
      if (existsById(db, 'consent_record', c.id)) continue;
      const entry = appendWithinTx(db, { ts: now, actor, action: 'CONSENT_RECORDED', payloadHash: computePayloadHash(c) });
      insertConsentRecord(db, c, entry.seq);
      outcome.consentsRecorded++;
    }
    for (const m of bundle.memos) {
      if (existsById(db, 'memo', m.id)) continue;
      const entry = appendWithinTx(db, { ts: now, actor, action: 'CREATE_MEMO', payloadHash: computePayloadHash(m) });
      insertMemo(db, m, entry.seq);
      outcome.memosStored++;
    }
    for (const r of resolved) {
      if (!r.sourcePath || r.actualSha === undefined) {
        insertInboxItem(db, {
          id: `inbox-${randomUUID()}`,
          sourcePath: r.ref.filename,
          detectedAt: now,
          sha256: r.ref.sha256,
          suggestedEventId: undefined,
          suggestedEncounterId: r.ref.encounterId,
          status: 'pending',
        });
        outcome.pendingMedia.push(r.ref.filename);
        continue;
      }
      if (r.actualSha !== r.ref.sha256) {
        insertInboxItem(db, {
          id: `inbox-${randomUUID()}`,
          sourcePath: r.ref.filename,
          detectedAt: now,
          sha256: r.actualSha,
          suggestedEncounterId: r.ref.encounterId,
          status: 'quarantined',
        });
        outcome.quarantinedMedia.push(r.ref.filename);
        continue;
      }
      const ingested = ingestArtifactCore(db, paths, {
        sourcePath: r.sourcePath,
        sha256: r.actualSha,
        size: statSync(r.sourcePath).size,
        mime: r.ref.mime,
        type: r.ref.type,
        capturedAt: r.ref.capturedAt,
        gps: r.ref.gps,
        deviceId: bundle.deviceId,
        encounterId: r.ref.encounterId,
        actor,
      });
      if (ingested) outcome.artifactsIngested.push(ingested.artifactId);
    }
    return outcome;
  })();
}
```

4. **错误语义分型**：`VaultKeyError`（口令错误 vs vault 损坏，openVault 用 sqlite_master 探测判别）、bundle 媒体三分态（ingested / pending-缺失 / quarantined-哈希不符，落 inbox_item）、原始件 temp+rename+chmod 0444、`pnpm-lock.yaml` 随依赖变更必须提交。
5. **测试环境与原生 ABI**：若沿用 better-sqlite3-multiple-ciphers 13 的 N-API 预编译（Plan 2 的选型，Node/Electron 通用），则无需 Electron-ABI 专链，普通 vitest 即可——Plan 2 的思路成立；本计划的 `ELECTRON_RUN_AS_NODE` 链路仅在使用 Electron ABI 专版时才需要。
